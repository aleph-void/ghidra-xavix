# XaviX p-code cases.  Expectations are worked out by hand from the MCS6500 Family Programming
# Manual (1976) and the published behaviour of the NMOS 6502 and its unintended opcodes.  The
# XaviX-specific parts (CALLF/RETF, banks) have no public document: their cases check the contract
# the module itself documents (bank byte pushed on the bank stack, return address pushed as
# next-1) and are marked "self-consistency".  Code is placed at extbus:8000.  Memory is zero unless
# set; the stack is lowbus:0100-01ff; S is the stack pointer low byte.

# ---------------- regression: a push must not touch the byte above the stack top ----------------
case jsr_does_not_clobber_caller_frame
code 200880a0074c0b80a20960
set S 0xFB
set lowbus:1FC 0x2E
set lowbus:1FD 0x82
expect lowbus:1FC 0x2E
expect lowbus:1FD 0x82
expect lowbus:1FB 0x80
expect lowbus:1FA 0x02
expect S 0xFB
end
case callf_does_not_clobber_caller_frame
code 22000980a0074c0c80a20980
set S 0xFB
set SSP 0xFF
set lowbus:1FC 0x2E
expect lowbus:1FC 0x2E
expect lowbus:1FB 0x80
expect lowbus:1FA 0x03
expect S 0xFB
end
case brk_does_not_clobber_caller_frame
code 00

expect lowbus:1FA 0x02

end
